// MNC Clean Architecture + Bloc Event State
class MobileAppBloc extends Bloc<AppEvent, AppState> {
  final IEnterpriseRepository _repository;
  
  MobileAppBloc({required IEnterpriseRepository repository})
      : _repository = repository,
        super(const AppState.initial()) {
    on<LoadProjectsEvent>(_onLoadProjects);
  }

  Future<void> _onLoadProjects(
    LoadProjectsEvent event,
    Emitter<AppState> emit,
  ) async {
    emit(const AppState.loading());
    final result = await _repository.getPublishedApps();
    result.fold(
      (failure) => emit(AppState.failure(failure)),
      (apps) => emit(AppState.success(apps)),
    );
  }
}
